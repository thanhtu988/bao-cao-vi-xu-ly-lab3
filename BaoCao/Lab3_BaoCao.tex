\subsection*{Exercise 3 -- Create STM32 Project}

\textbf{Yêu cầu:} tạo project có chân đúng theo sơ đồ, ngắt timer khoảng 10 ms.

GPIO Output: Push Pull, mức ban đầu \textbf{High} (đèn tắt). GPIO Input PB12--PB14: No pull.
TIM2: Internal Clock, PSC = 7999, ARR = 9, bật NVIC.

\begin{equation*}
f_{\text{ngắt}} = \frac{8\,000\,000}{(7999+1)\times(9+1)} = 100\ \text{Hz}
\;\Rightarrow\; T_{\text{ngắt}} = 10\ \text{ms}
\end{equation*}

\begin{lstlisting}[caption={Exercise 3 -- main.c (Program 1.9)}]
  MX_GPIO_Init();
  MX_TIM2_Init();
  /* USER CODE BEGIN 2 */
  HAL_TIM_Base_Start_IT(&htim2);
  /* USER CODE END 2 */

  /* Infinite loop */
  /* USER CODE BEGIN WHILE */
  while (1)
  {
    fsm_for_input_processing();
    /* USER CODE END WHILE */

    /* USER CODE BEGIN 3 */
  }
  /* USER CODE END 3 */
\end{lstlisting}

% ---------------- EX 4 ----------------
\subsection*{Exercise 4 -- Modify Timer Parameters}

\textbf{Yêu cầu:} khi đổi chu kì ngắt (1 ms, 100 ms) thì sửa ít nhất và không ảnh hưởng hệ thống (đèn vẫn 2 Hz).

\textbf{Giải pháp:} mọi thời gian đều viết bằng \textbf{ms}, chỉ đổi ra số nhịp ngắt ở một chỗ là
\texttt{setTimerX()} (chia cho \texttt{TIMER\_CYCLE}). Đổi chu kì ngắt chỉ cần sửa PSC/ARR và \texttt{TIMER\_CYCLE}.

\begin{lstlisting}[caption={Exercise 4 -- timer.h, timer.c}]
/* Chu ky ngat cua TIM2, don vi ms.
   Exercise 4: khi doi chu ky ngat (PSC/ARR cua TIM2 trong file .ioc)
   chi can sua DUY NHAT hang so nay. Moi khoang thoi gian khac trong
   chuong trinh deu viet bang ms va duoc doi ra so nhip ngat qua TIMER_CYCLE. */
#define TIMER_CYCLE   10

void setTimer0(int duration)
{
  timer0_counter = duration / TIMER_CYCLE;
  timer0_flag    = 0;
}

void HAL_TIM_PeriodElapsedCallback(TIM_HandleTypeDef *htim)
{
  if (htim->Instance == TIM2) {
    button_reading();
    timer_run();
  }
}
\end{lstlisting}

\begin{lstlisting}[caption={Exercise 4 -- cac thoi gian viet bang ms (input\_processing.c)}]
#define ONE_SECOND        1000          /* nhip dem lui cua den giao thong     */
#define BLINK_DURATION    250           /* 2 Hz: sang 250 ms + tat 250 ms = 0,5 s */
#define SCAN_DURATION     TIMER_CYCLE   /* moi nhip ngat quet 1 LED 7 doan     */
\end{lstlisting}

\begin{center}
\small
\begin{tabular}{|c|c|c|c|c|}
\hline
\textbf{Chu kì ngắt} & \textbf{PSC} & \textbf{ARR} & \texttt{TIMER\_CYCLE} & \textbf{Đèn 2 Hz (250 ms)} \\\hline
1 ms   & 799  & 9  & 1   & 250 nhịp -- đúng \\\hline
10 ms  & 7999 & 9  & 10  & 25 nhịp -- đúng \\\hline
100 ms & 7999 & 99 & 100 & 2 nhịp = 200 ms \\\hline
\end{tabular}
\end{center}

\begin{ghichu}
\textbf{2 Hz} = một chu kì sáng--tắt 0,5 s $\Rightarrow$ đảo trạng thái mỗi \textbf{250 ms}.
Ở 100 ms, \texttt{250/100} chia nguyên còn 2 nhịp nên đèn đảo mỗi 200 ms: software timer không đo được
khoảng lẻ hơn chu kì ngắt. Với 1 ms phải dùng PSC = 799, ARR = 9 (không dùng ARR = 0 như đã gặp ở Lab 2).
\end{ghichu}

void button_reading(void)
{
  for (uint8_t i = 0; i < N0_OF_BUTTONS; i++) {
    debounceButtonBuffer2[i] = debounceButtonBuffer1[i];
    debounceButtonBuffer1[i] = HAL_GPIO_ReadPin(buttonPort[i], buttonPin[i]);
    if (debounceButtonBuffer1[i] == debounceButtonBuffer2[i])
      buttonBuffer[i] = debounceButtonBuffer1[i];
    if (buttonBuffer[i] == BUTTON_IS_PRESSED) {
      // if a button is pressed, we start counting
      if (counterForButtonPress1s[i] < DURATION_FOR_AUTO_INCREASING) {
        counterForButtonPress1s[i]++;
      } else {
        // the flag is turned on when 1 second has passed
        // since the button is pressed.
        flagForButtonPress1s[i] = 1;
      }
    } else {
      counterForButtonPress1s[i] = 0;
      flagForButtonPress1s[i] = 0;
    }
  }
}
\end{lstlisting}

static unsigned char fsm_for_button(unsigned char index)
{
  unsigned char pressed = 0;

  switch (buttonState[index]) {
  case BUTTON_RELEASED:
    if (is_button_pressed(index)) {
      buttonState[index] = BUTTON_PRESSED;
      pressed = 1;
    }
    break;
  case BUTTON_PRESSED:
    if (!is_button_pressed(index)) {
      buttonState[index] = BUTTON_RELEASED;
    } else {
      if (is_button_pressed_1s(index)) {
        buttonState[index] = BUTTON_PRESSED_MORE_THAN_1_SECOND;
      }
    }
    break;
  case BUTTON_PRESSED_MORE_THAN_1_SECOND:
    if (!is_button_pressed(index)) {
      buttonState[index] = BUTTON_RELEASED;
    }
    /* de bai khong yeu cau tu tang khi giu nut -> khong lam gi them */
    break;
  }
  return pressed;
}
\end{lstlisting}

static void traffic_tick(int road)
{
  light_counter[road]--;
  if (light_counter[road] <= 0) {
    switch (light_color[road]) {
    case COLOR_RED:
      light_color[road]   = COLOR_GREEN;
      light_counter[road] = duration_green;
      break;
    case COLOR_GREEN:
      light_color[road]   = COLOR_AMBER;
      light_counter[road] = duration_amber;
      break;
    case COLOR_AMBER:
      light_color[road]   = COLOR_RED;
      light_counter[road] = duration_red;
      break;
    default:
      break;
    }
    setTrafficLight(road, light_color[road]);
  }
}
